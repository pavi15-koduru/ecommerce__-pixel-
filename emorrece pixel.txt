import 'package:flutter/material.dart';

void main() {
  runApp(const OverflowAvoidanceApp());
}

class OverflowAvoidanceApp extends StatelessWidget {
  const OverflowAvoidanceApp({super.key});

  @override
  Widget build(BuildContext context) {
    return const MaterialApp(
      home: Q2KeyboardOverflowScreen(),
    );
  }
}

class Q2KeyboardOverflowScreen extends StatelessWidget {
  const Q2KeyboardOverflowScreen({super.key});

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(
        title: const Text('Q2: Avoid Bottom Overflow (30px)'),
      ),
      // **SingleChildScrollView: Correctly prevents keyboard overflow**
      body: SingleChildScrollView(
        padding: const EdgeInsets.all(20.0),
        child: Column(
          children: <Widget>[
            // Placeholder content to take up space
            Container(height: 300, color: Colors.grey[200], child: const Center(child: Text('Header Content'))),
            
            const SizedBox(height: 50),
            
            const Text(
              'Scroll down and tap the input field to test keyboard behavior.',
              textAlign: TextAlign.center,
              style: TextStyle(fontSize: 16),
            ),
            
            // Large empty space to push the TextField near the bottom
            const SizedBox(height: 350), 
            
            // The TextField that will cause overflow if not wrapped
            const TextField(
              decoration: InputDecoration(
                labelText: 'User Input Field',
                border: OutlineInputBorder(),
              ),
            ),
            
            const SizedBox(height: 20),
            
            // Another element that would be clipped without scrolling
            ElevatedButton(
              onPressed: () {},
              child: const Text('Submit Form'),
            ),
          ],
        ),
      ),
    );
  }
}
